            method: 'GET', 
            url: CONFIG.CLUSTER_COMMUNICATOR+'lock/walletlock/'+xHash

          if(response.data=="found") callback(true);
          else callback(false);
        }).catch(function (error) {
          //console.error(error);
          callback(false);
        });

isProductTypeSettlement: function(product, subproduct, callback) {
    dbRead_selectQ("products", "*", {
        "prod_code": product,
    }, {}, function (productData, errMsg) {
        console.log('productData - ', productData)
        if (!productData || productData[0] == null) {
            callback(false, false);
            return;
        }
        if (productData[0]['is_settlement'] == "true") {
            callback(true, productData[0], false);
            return;
        } else {
            //CHECK SUBPRODUCT
            if (subproduct) {
                dbRead_selectQ("sys_subproducts", "*", {
                    "product": product,
                    "subproduct": subproduct,
                }, {}, function (subproductData, errMsg) {
                    console.log('subproductData 3- ', subproductData, errMsg)
                    if (!subproductData || subproductData[0] == null) {
                        callback(false, productData[0]);
                        return;
                    } else {
                        if (subproductData[0]['is_settlement'] == "true") {
                            callback(true, productData[0]);
                            return;
                        } else {
                            callback(false, productData[0]);
                            return;
                        }
                    }
                });
            } else {
                callback(false, productData[0]);
            }
        }
    });
}

findSchemeAuto: function (productData, dataParams, callback) {

        console.log('tdsValue 2 - ', tdsValue)

findScheme2: function(accID, productCode, subproductCode, walletType, callback) {
    var whereClause = {
        "schemes_allocated.accountids_id": accID,//+"X"
        "schemes.scheme_product": productCode,
        "schemes.scheme_subproduct": subproductCode,
        "products.blocked": "false",
        "schemes.blocked": "false",
        "schemes_allocated.blocked": "false",
        "schemes.id=schemes_allocated.scheme_id": "RAW",
        "schemes.scheme_product=products.prod_code": "RAW"
    };
    if (walletType != null) {
        whereClause["FIND_IN_SET('" + walletType + "', schemes.wallet_type)"] = "RAW";
    }
    dbRead_selectQ("schemes,schemes_allocated,products", "schemes.*,products.*,schemes.id as schemeID,products.id as productID", whereClause, {}, function (schemeData) {
        if (!schemeData) {
            SCHEME.getFallbackSchemeMap(accID, productCode, walletType, callback);
            return;
        }
        schemeData = schemeData[0];
        var schemeDefination = {};
        try {
            schemeDefination = JSON.parse(schemeData.scheme_defination);
            if (!schemeDefination) {
                return callback(false, "Scheme Is Corrupted, please check with admin");
            }
        } catch (e) {
            return callback(false, "Scheme Is Corrupted or not defined, please check with admin");
        }
        schemeData.scheme_defination = schemeDefination;
        callback(schemeData);
    });
}

fetchSchemeSlab: function(schemeData, txamount, operatorCode, callback) {
    var schemeDefination = schemeData.scheme_defination;
    var slabArr = [];
    if (schemeData['operator_source'] != null && schemeData['operator_source'].length > 0) {
        if (schemeDefination.slabs[operatorCode] != null) {
            slabArr = schemeDefination.slabs[operatorCode];
        } else if (schemeDefination.slabs[operatorCode.toLowerCase()] != null) {
            slabArr = schemeDefination.slabs[operatorCode.toLowerCase()];
        } else {
            //###
            console.log("Operator Not Supported", { "operator_source": schemeData['operator_source'] }, { "operatorCode": operatorCode }, { "schemeDefination": schemeDefination }, { "schemeData": schemeData });
            //###
            return callback(false, "Operator Not Supported", { "operator": operatorCode });
        }
    } else {
        slabArr = schemeDefination.slabs;
    }
    var txAmount = parseFloat(txamount);
    if (schemeData['cluster_mode'] == "true") {
        var clusterAmount = parseFloat(schemeData['cluster_amount']);
        if (clusterAmount < txAmount) {
            return callback(false, "Cluster Mode Is Not Supported Yet");
        } else {
            var txSlab = this.indentifySlab(slabArr, txamount);
            if (!txSlab) {
                return callback(false, "TX Slab could not be identified");
            }
            callback(txSlab);
        }
    } else {
        var txSlab = this.indentifySlab(slabArr, txamount);
        if (!txSlab) {
            return callback(false, "TX Slab could not be identified");
        }
        callback(txSlab);
    }
}

indentifySlab: function(slabArr, txAmount) {
	var txSlab = false;
	var txAmount = parseInt(txAmount);
	var slabKeys = Object.keys(slabArr);
	var keyLength = slabKeys.length;
	for (i = 0; i < keyLength; i++) {
		slab = slabArr[slabKeys[i]];
		slab.slab_start = parseInt(slab.slab_start);
		slab.slab_end = parseInt(slab.slab_end);
		if (txAmount >= slab.slab_start && txAmount <= slab.slab_end) {
			txSlab = slab;
		}
	}
	return txSlab;
}
